"""Read-only local dashboard for existing Corridor Lifeline snapshots."""
import argparse
from datetime import datetime, timezone
from html import escape
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
import json
import math
from pathlib import Path
from urllib.parse import parse_qs, urlsplit

ROOT = Path(__file__).resolve().parent.parent
VIEWS = {'live': ('Live snapshot', Path('live/latest.json')),
         '2022': ('Historical demo · 2022', Path('demo/2022/latest.json')),
         '2024': ('Historical demo · 2024', Path('demo/2024/latest.json'))}
MAX_BYTES = 2 * 1024 * 1024


def timestamp(value):
    if not isinstance(value, str):
        raise ValueError('Snapshot time must be a timezone-aware timestamp')
    result = datetime.fromisoformat(value.replace('Z', '+00:00'))
    if result.tzinfo is None:
        raise ValueError('Snapshot time must include a timezone')
    return result


def read_snapshot(root, view):
    """Only fixed snapshot paths are read; no inputs are collected or written."""
    path = Path(root) / VIEWS[view][1]
    try:
        with path.open('rb') as stream:
            content = stream.read(MAX_BYTES + 1)
        if len(content) > MAX_BYTES:
            raise ValueError('Snapshot exceeds the 2 MB limit')
        data = json.loads(content)
        if not isinstance(data, dict):
            raise ValueError('Snapshot must be an object')
        required = ('corridor', 'as_of', 'rule_version', 'level', 'rail_status', 'coverage')
        if any(not isinstance(data.get(key), str) for key in required):
            raise ValueError('Snapshot is missing required text fields')
        timestamp(data['as_of'])
        score = data.get('score')
        if isinstance(score, bool) or not isinstance(score, (int, float)) or not math.isfinite(score) or not 0 <= score <= 100:
            raise ValueError('Snapshot score must be between 0 and 100')
        if not isinstance(data.get('components'), dict) or any(isinstance(v, bool) or not isinstance(v, (int, float)) or not math.isfinite(v) or not 0 <= v <= 100 for v in data['components'].values()):
            raise ValueError('Snapshot components must contain scores between 0 and 100')
        for key in ('reasons', 'issues', 'sources'):
            if not isinstance(data.get(key), list) or any(not isinstance(v, str) for v in data[key]):
                raise ValueError(f'Snapshot {key} must be a list of text')
        if not isinstance(data.get('feeds'), list) or any(not isinstance(v, dict) for v in data['feeds']):
            raise ValueError('Snapshot feeds must be a list of objects')
        return data, None
    except FileNotFoundError:
        return None, 'No snapshot available. Generate an offline demo or run a collection separately, then refresh this page.'
    except (OSError, ValueError, TypeError, UnicodeError) as exc:
        return None, f'Snapshot cannot be read: {exc}'


def freshness(data, now, max_age_minutes):
    age = (now - timestamp(data['as_of'])).total_seconds() / 60
    if age < -5:
        return 'future', 'Future-dated snapshot: its collection time is more than five minutes ahead. Current status is unavailable.'
    if age > max_age_minutes:
        return 'stale', f'Stale snapshot: collected {age:,.0f} minutes ago. Current status is unavailable.'
    return 'fresh', f'Snapshot age: {max(0, age):.0f} minutes. Source health and verification times still determine coverage.'


def text(value):
    return escape(str(value), quote=True)


def link(value):
    """Link public HTTP(S) sources only; render local/unsafe references as text."""
    value = str(value)
    parsed = urlsplit(value)
    if parsed.scheme.lower() in ('http', 'https') and parsed.netloc and not any(ord(c) < 32 for c in value):
        return f'<a href="{text(value)}" target="_blank" rel="noopener noreferrer">{text(value)}</a>'
    return text(value)


def listing(values):
    return '<ul>' + ''.join(f'<li>{text(v)}</li>' for v in values) + '</ul>' if values else '<p class="muted">None recorded.</p>'


STYLE = '''*{box-sizing:border-box}body{margin:0;background:#0d1421;color:#e7edf7;font:16px/1.6 system-ui,sans-serif}main{max-width:1150px;margin:auto;padding:32px 24px}h1{font-size:32px;line-height:1.2}h2{font-size:18px;margin:0 0 14px}a{color:#83d6df;overflow-wrap:anywhere}nav{display:flex;gap:12px;flex-wrap:wrap;margin:24px 0}nav a{padding:9px 16px;border:1px solid #35455c;border-radius:24px;text-decoration:none}nav a.active{background:#1b5961;color:white}.eyebrow{letter-spacing:.15em;color:#83d6df;font-size:12px}.muted{color:#afbdd0}.banner{background:#4b3615;border:1px solid #bb8932;padding:18px;border-radius:12px;margin:18px 0}.grid{display:grid;grid-template-columns:1fr 1fr;gap:18px}.card{background:#172337;border:1px solid #2b3b52;border-radius:16px;padding:24px;overflow-wrap:anywhere}.score{font-size:60px;line-height:1.2;font-weight:700}.pill{color:#8ee1cb}li{margin:8px 0}dl{display:grid;grid-template-columns:145px 1fr;gap:5px 12px}dt{color:#afbdd0}dd{margin:0;overflow-wrap:anywhere}pre{white-space:pre-wrap;font:13px/1.5 ui-monospace,monospace;background:#101a2a;padding:14px;border-radius:8px}footer{margin-top:24px;font-size:13px;color:#afbdd0}@media(max-width:700px){.grid{grid-template-columns:1fr}main{padding:24px 16px}dl{grid-template-columns:1fr}h1{font-size:27px}}'''


def render(root, view='live', now=None, max_age_minutes=30):
    now = now or datetime.now(timezone.utc)
    data, error = read_snapshot(root, view)
    nav = ''.join(f'<a class="{"active" if key == view else ""}" href="/?view={key}">{label}</a>' for key, (label, _) in VIEWS.items())
    body = '<p class="eyebrow">ATLASIQ / CORRIDOR LIFELINE</p><h1>East-west corridor intelligence</h1><p class="muted">Explainable monitoring, with source gaps in view.</p><nav>' + nav + '</nav>'
    if view != 'live':
        body += '<div class="banner"><strong>HISTORICAL DEMO — NEVER LIVE</strong><br>Weather and warnings are illustrative. This replay is not a predictive backtest.</div>'
    if error:
        body += f'<section class="card"><h2>Snapshot unavailable</h2><p>{text(error)}</p><p>Coverage and current rail status: UNKNOWN.</p></section>'
    else:
        state, message = freshness(data, now, max_age_minutes)
        historical = view != 'live' or 'fixture_note' in data
        if historical and view == 'live':
            body += '<div class="banner"><strong>HISTORICAL DATA IN LIVE LOCATION — NEVER LIVE</strong><br>This snapshot contains fixture provenance and cannot establish current conditions.</div>'
        if not historical:
            body += f'<div class="banner"><strong>{text(state.upper())} SNAPSHOT</strong><br>{text(message)}<br>Age policy: {max_age_minutes:g} minutes; future tolerance: 5 minutes.</div>'
        current = not historical and state == 'fresh'
        prefix = 'Historical' if historical else 'Recorded'
        body += f'<p>{text(data["corridor"])}</p><div class="grid"><section class="card"><h2>{prefix} rule score</h2><div class="score">{text(data["score"])}<span class="muted" style="font-size:24px"> / 100</span></div><p class="pill">{text(data["level"])} · rule v{text(data["rule_version"])}</p><p class="muted">A rule score, not a probability or closure prediction.</p>'
        body += listing([f'{key}: {value} points' for key, value in data['components'].items()]) + '</section>'
        body += f'<section class="card"><h2>Coverage & rail status</h2><p><strong>{text(data["coverage"]) if current else "Current coverage unavailable"}</strong></p><p>{"Reported rail status" if current else prefix + " rail status"}: {text(data["rail_status"])}</p><p>Collected: {text(data["as_of"])}</p><p class="muted">Complete coverage refers only to configured inputs. Fetching does not renew rail verification.</p><h2>Recorded gaps</h2>' + listing(data['issues']) + '</section>'
        body += '<section class="card"><h2>Why this score</h2>' + listing(data['reasons']) + '</section><section class="card"><h2>Sources & attribution</h2><ul>' + ''.join(f'<li>{link(v)}</li>' for v in data['sources']) + '</ul><p class="muted">Weather and warnings: Bureau of Meteorology. Rail: manually verified operator announcements. Source permissions require review before commercial use.</p></section></div><h2 style="margin-top:28px">Feed evidence</h2><div class="grid">'
        for feed in data['feeds']:
            detail = feed.get('data', {})
            detail = detail if isinstance(detail, dict) else {}
            body += '<section class="card"><h2>' + text(feed.get('name', 'Unnamed feed')) + '</h2><dl>'
            for name, value in [('Kind', feed.get('kind', 'unknown')), ('Recorded health', feed.get('health', 'unknown')), ('Fetched at', feed.get('fetched_at', 'Missing')), ('Observed / verified', detail.get('observed_at', 'Missing')), ('Error', feed.get('error', 'None recorded'))]:
                body += f'<dt>{text(name)}</dt><dd>{text(value)}</dd>'
            body += '<dt>Source</dt><dd>' + link(feed.get('source', 'Missing')) + '</dd></dl><details><summary>Recorded input detail</summary><pre>' + text(json.dumps(detail, indent=2, ensure_ascii=False)) + '</pre></details></section>'
        body += '</div>'
        if historical:
            body += '<section class="card" style="margin-top:18px"><h2>Historical provenance</h2><p>' + text(data.get('fixture_note', 'Historical demo snapshot.')) + '</p>'
            facts = data.get('event_facts', [])
            if isinstance(facts, list):
                for fact in facts:
                    if isinstance(fact, dict):
                        body += '<p>' + text(fact.get('fact', '')) + ' · ' + link(fact.get('source', '')) + '</p>'
            body += '</section>'
    body += '<footer>Read-only local viewer. Refreshing reads saved output; it does not collect data or send alerts. Page refreshes every 60 seconds. Times retain their source timezone.</footer>'
    return '<!doctype html><html lang="en"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"><meta http-equiv="refresh" content="60"><title>AtlasIQ · Corridor Lifeline</title><style>' + STYLE + '</style></head><body><main>' + body + '</main></body></html>'


def handler(output_root, max_age_minutes=30):
    class DashboardHandler(BaseHTTPRequestHandler):
        def do_GET(self):
            url = urlsplit(self.path)
            view = parse_qs(url.query).get('view', ['live'])[0]
            if url.path != '/' or view not in VIEWS:
                self.send_error(404, 'Unknown dashboard view')
                return
            content = render(output_root, view, max_age_minutes=max_age_minutes).encode('utf-8')
            self.send_response(200)
            self.send_header('Content-Type', 'text/html; charset=utf-8')
            self.send_header('Content-Length', str(len(content)))
            self.send_header('Cache-Control', 'no-store')
            self.send_header('Content-Security-Policy', "default-src 'none'; style-src 'unsafe-inline'; base-uri 'none'; frame-ancestors 'none'")
            self.send_header('X-Content-Type-Options', 'nosniff')
            self.end_headers()
            self.wfile.write(content)
    return DashboardHandler


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--out', type=Path, default=ROOT / 'output', help='Existing collection output root')
    parser.add_argument('--port', type=int, default=8765)
    parser.add_argument('--max-age-minutes', type=float, default=30)
    args = parser.parse_args(argv)
    if not 0 <= args.port <= 65535:
        parser.error('--port must be between 0 and 65535')
    if not math.isfinite(args.max_age_minutes) or args.max_age_minutes <= 0:
        parser.error('--max-age-minutes must be finite and positive')
    try:
        with ThreadingHTTPServer(('127.0.0.1', args.port), handler(args.out, args.max_age_minutes)) as server:
            print(f'AtlasIQ dashboard: http://127.0.0.1:{server.server_port}/ (read-only)', flush=True)
            server.serve_forever()
    except KeyboardInterrupt:
        return 0
    except OSError as exc:
        print(f'Dashboard could not start: {exc}')
        return 1
    return 0


if __name__ == '__main__':
    raise SystemExit(main())
